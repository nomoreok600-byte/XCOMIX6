@'
require('dotenv').config();
const { launch, getStream } = require('puppeteer-stream');
const { spawn } = require('child_process');

const CONFIG = {
  url: process.env.TARGET_WEBSITE,
  streamKey: process.env.YOUTUBE_KEY,
  width: 1280,
  height: 720,
  fps: 30
};

async function startStream() {
  console.log('1. Starting browser...');
  const browser = await launch({
    headless: false, 
    defaultViewport: { width: CONFIG.width, height: CONFIG.height },
    args: [
      '--window-size=' + CONFIG.width + ',' + CONFIG.height,
      '--autoplay-policy=no-user-gesture-required',
      '--no-sandbox'
    ]
  });

  const page = await browser.newPage();
  console.log('2. Opening game website...');
  await page.goto(CONFIG.url, { waitUntil: 'networkidle2' });

  console.log('3. Capturing stream...');
  const webStream = await getStream(page, {
    audio: true,
    video: true,
    frameSize: 20,
    mimeType: 'video/webm;codecs=vp8,opus'
  });

  console.log('4. Broadcasting to YouTube...');
  const ffmpeg = spawn('ffmpeg', [
    '-i', '-',
    '-c:v', 'libx264',
    '-preset', 'ultrafast',
    '-r', CONFIG.fps.toString(),
    '-b:v', '3500k',
    '-pix_fmt', 'yuv420p',
    '-c:a', 'aac',
    '-b:a', '128k',
    '-f', 'flv',
    'rtmp://a.rtmp.youtube.com/live2/' + CONFIG.streamKey
  ]);

  webStream.pipe(ffmpeg.stdin);

  ffmpeg.on('close', () => {
    console.log('Stream disconnected. Restarting...');
    browser.close();
    setTimeout(startStream, 5000);
  });
}

startStream();
