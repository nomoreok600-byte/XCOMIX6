$ProgressPreference = 'SilentlyContinue';
Set-Location "C:\auto-stream"

$code = @(
"const puppeteer = require('puppeteer');",
"const { launch, getStream } = require('puppeteer-stream');",
"const { spawn } = require('child_process');",
"const CONFIG = {",
"  url: 'https://stream.virallinx.site/',",
"  streamKey: 'm3kx-5ae5-k3br-xr31-6h0w',",
"  width: 1280,",
"  height: 720,",
"  fps: 30",
"};",
"async function startStream() {",
"  console.log('1. Locating Chrome...');",
"  const chromePath = await puppeteer.executablePath();",
"  console.log('2. Starting browser...');",
"  const browser = await launch({",
"    headless: false,",
"    executablePath: chromePath,",
"    defaultViewport: { width: CONFIG.width, height: CONFIG.height },",
"    args: ['--window-size=1280,720', '--autoplay-policy=no-user-gesture-required', '--no-sandbox']",
"  });",
"  const page = await browser.newPage();",
"  page.setDefaultNavigationTimeout(0);",
"  console.log('3. Opening game website...');",
"  await page.goto(CONFIG.url, { waitUntil: 'domcontentloaded' });",
"  console.log('4. Capturing stream...');",
"  const webStream = await getStream(page, { audio: true, video: true, frameSize: 20, mimeType: 'video/webm;codecs=vp8,opus' });",
"  console.log('5. Broadcasting to YouTube...');",
"  const ffmpeg = spawn('ffmpeg', ['-i', '-', '-c:v', 'libx264', '-preset', 'ultrafast', '-r', '30', '-b:v', '3500k', '-pix_fmt', 'yuv420p', '-c:a', 'aac', '-b:a', '128k', '-f', 'flv', 'rtmp://a.rtmp.youtube.com/live2/' + CONFIG.streamKey]);",
"  webStream.pipe(ffmpeg.stdin);",
"  ffmpeg.on('close', () => { console.log('Stream disconnected. Restarting...'); browser.close(); setTimeout(startStream, 5000); });",
"}",
"startStream();"
)
$code | Out-File -FilePath "C:\auto-stream\stream.js" -Encoding utf8

Write-Host "Launching fixed stream..." -ForegroundColor Cyan
node stream.js
